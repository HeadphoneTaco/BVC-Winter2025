// Create a class named 'Dice'
    class Dice
    {
        // Create new instance of class named 'Random', with 'RandomRoll' as a field
        static Random RandomRoll = new();
        
        // Has to be static, because ''''reasons''''
        public static int Roll(int sides)
        {
            int roll = RandomRoll.Next(1, sides + 1);
            // Randomizes a number between 1 and the number of sides
            return roll;
        }
    }

// Create a class named 'PerformCheck'
    class PerformCheck
    {
        bool DexCheck()
        {
            int dexterityCheck = Dice.Roll(20) + (int)SkillModifier.Dexterity;
            return dexterityCheck >= 10;
        }
        bool StrengthCheck()
        {
            int strengthCheck = Dice.Roll(20) + (int)SkillModifier.Strength;
            return strengthCheck >= 10;
        }
        bool CharismaCheck()
        {
            int charismaCheck = Dice.Roll(20) + (int)SkillModifier.Charisma;
            return charismaCheck >= 10;
        }
        bool AttackCheck()
        {
            int attackCheck = Dice.Roll(20) + (int)SkillModifier.Attack;
            return attackCheck >= 10;
        }
        bool WisdomCheck()
        {
            int wisdomCheck = Dice.Roll(20) + (int)SkillModifier.Wisdom;
            return wisdomCheck >= 10;
        }
        bool IntelligenceCheck()
        {
            int intelligenceCheck = Dice.Roll(20) + (int)SkillModifier.Intelligence;
            return intelligenceCheck >= 10;
        }
        bool ConstitutionCheck()
        {
            int constitutionCheck = Dice.Roll(20) + (int)SkillModifier.Constitution;
            return constitutionCheck >= 10;
        }
    }
